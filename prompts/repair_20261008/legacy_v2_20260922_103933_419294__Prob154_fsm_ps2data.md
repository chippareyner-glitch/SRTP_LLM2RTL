## 本批失败RTL

```systemverilog
module TopModule (
    input  logic        clk,
    input  logic        reset,
    input  logic [7:0]  in,
    output logic [23:0] out_bytes,
    output logic        done
);

    // FSM state definitions
    localparam [1:0] IDLE   = 2'b00,
                     SECOND = 2'b01,
                     THIRD  = 2'b10,
                     DONE   = 2'b11;

    logic [1:0] state;
    logic [7:0] byte1, byte2, byte3;

    always_ff @(posedge clk) begin
        if (reset) begin
            state     <= IDLE;
            byte1     <= 8'h00;
            byte2     <= 8'h00;
            byte3     <= 8'h00;
            out_bytes <= 24'h000000;
            done      <= 1'b0;
        end else begin
            // done is asserted in the DONE state, which is reached
            // one cycle after the third byte has been received.
            done <= (state == DONE);

            case (state)
                IDLE: begin
                    // Wait for the first byte of a message.
                    // A start byte is identified by bit[3] = 1.
                    if (in[3]) begin
                        byte1 <= in;
                        state <= SECOND;
                    end
                end

                SECOND: begin
                    // Second byte of the message.
                    byte2 <= in;
                    state <= THIRD;
                end

                THIRD: begin
                    // Third byte of the message.
                    byte3 <= in;
                    state <= DONE;
                end

                DONE: begin
                    // Pack the three received bytes.
                    out_bytes <= {byte1, byte2, byte3};

                    // The byte present during this cycle is the first
                    // candidate for the next message. If it qualifies,
                    // start a new message immediately; otherwise return
                    // to discarding bytes.
                    if (in[3]) begin
                        byte1 <= in;
                        state <= SECOND;
                    end else begin
                        state <= IDLE;
                    end
                end

                default: state <= IDLE;
            endcase
        end
    end

endmodule
```
