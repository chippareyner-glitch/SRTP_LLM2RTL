```systemverilog
module TopModule (
    input  logic clk,
    input  logic areset,
    input  logic x,
    output logic z
);

typedef enum logic [1:0] {
    STATE_IDLE,
    STATE_PASS,
    STATE_INVERT
} state_t;

state_t current_state, next_state;
logic found_one;

// Sequential logic
always_ff @(posedge clk or posedge areset) begin
    if (areset) begin
        current_state <= STATE_IDLE;
        found_one <= 1'b0;
    end else begin
        current_state <= next_state;
        if (current_state == STATE_IDLE) begin
            found_one <= 1'b0;
        end else if (current_state == STATE_PASS && x) begin
            found_one <= 1'b1;
        end
    end
end

// Next state logic
always_comb begin
    case (current_state)
        STATE_IDLE: next_state = STATE_PASS;
        STATE_PASS: next_state = found_one ? STATE_INVERT : STATE_PASS;
        STATE_INVERT: next_state = STATE_INVERT;
        default: next_state = STATE_IDLE;
    endcase
end

// Output logic (Moore)
always_comb begin
    if (areset) begin
        z = 1'b0;  // Drive known value during reset
    end else begin
        case (current_state)
            STATE_IDLE: z = 1'b0;
            STATE_PASS: z = x;
            STATE_INVERT: z = ~x;
            default: z = 1'b0;
        endcase
    end
end

endmodule
```
